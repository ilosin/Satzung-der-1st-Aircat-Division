\section*{\S~11 Gültigkeit}
\label{sec:gueltigkeit}
\addcontentsline{toc}{section}{\S~11 Gültigkeit}

\begin{enumerate}[label=(\arabic*)]
    \item Diese Satzung verliert ihre Gültigkeit, wenn
    \begin{enumerate}[label=(\Alph*)]
        \item die Mitglieder der 1st Aircat Division sich in einer demokratischen Entscheidung ein neues, vergleichbares Dokument gegeben haben.
        \item die Mitglieder der 1st Aircat Division sich in einer demokratischen Entscheidung zur Auflösung der 1st Aircat Division entschieden haben.
    \end{enumerate}
    Hierbei sind die Voraussetzungen die gleichen wie nach \S~2e \ref{itm:satzungsaenderungen}.
\end{enumerate}
